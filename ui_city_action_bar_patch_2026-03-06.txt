func _ensure_city_action_bar() -> void:
    var root_vb := $Margin/VBox as VBoxContainer
    if root_vb == null:
        return

    city_action_bar = root_vb.get_node_or_null("CityActionBar") as HBoxContainer
    if city_action_bar == null:
        city_action_bar = HBoxContainer.new()
        city_action_bar.name = "CityActionBar"
        city_action_bar.alignment = BoxContainer.ALIGNMENT_BEGIN
        city_action_bar.add_theme_constant_override("separation", 6)
        city_action_bar.custom_minimum_size = Vector2(0, 38)
        root_vb.add_child(city_action_bar)

        if is_instance_valid(topbar) and city_action_bar.get_parent() == root_vb:
            root_vb.move_child(city_action_bar, topbar.get_index() + 1)

    city_trade_btn = _ensure_city_action_button(city_action_bar, "CityTradeBtn", "Trade")
    city_map_btn = _ensure_city_action_button(city_action_bar, "CityMapBtn", "Map")
    city_step_btn = _ensure_city_action_button(city_action_bar, "CityStepBtn", "+1 Day")
    city_step_turn_btn = _ensure_city_action_button(city_action_bar, "CityStepTurnBtn", "+1 Turn")
    city_move_btn = _ensure_city_action_button(city_action_bar, "CityMoveBtn", "Move")
    city_contract_btn = _ensure_city_action_button(city_action_bar, "CityContractBtn", "契約")
    city_info_btn = _ensure_city_action_button(city_action_bar, "CityInfoBtn", "情報")
    city_inv_btn = _ensure_city_action_button(city_action_bar, "CityInvBtn", "Inv")
    city_trust_btn = _ensure_city_action_button(city_action_bar, "CityTrustBtn", "信用")
    city_maint_btn = _ensure_city_action_button(city_action_bar, "CityMaintBtn", "整備")

func _ensure_city_action_button(parent: HBoxContainer, name: String, text: String) -> Button:
    var btn := parent.get_node_or_null(name) as Button
    if btn == null:
        btn = Button.new()
        btn.name = name
        btn.text = text
        btn.custom_minimum_size = Vector2(92, 32)
        btn.size_flags_horizontal = Control.SIZE_SHRINK_CENTER
        btn.mouse_filter = Control.MOUSE_FILTER_STOP
        btn.focus_mode = Control.FOCUS_NONE
        parent.add_child(btn)
    return btn

func _wire_city_action_buttons() -> void:
    if city_trade_btn and not city_trade_btn.pressed.is_connected(Callable(self, "_on_city_trade_btn")):
        city_trade_btn.pressed.connect(_on_city_trade_btn)
    if city_map_btn and not city_map_btn.pressed.is_connected(Callable(self, "_on_city_map_btn")):
        city_map_btn.pressed.connect(_on_city_map_btn)
    if city_step_btn and not city_step_btn.pressed.is_connected(Callable(self, "_on_city_step_btn")):
        city_step_btn.pressed.connect(_on_city_step_btn)
    if city_step_turn_btn and not city_step_turn_btn.pressed.is_connected(Callable(self, "_on_city_step_turn_btn")):
        city_step_turn_btn.pressed.connect(_on_city_step_turn_btn)
    if city_move_btn and not city_move_btn.pressed.is_connected(Callable(self, "_on_city_move_btn")):
        city_move_btn.pressed.connect(_on_city_move_btn)
    if city_contract_btn and not city_contract_btn.pressed.is_connected(Callable(self, "_on_city_contract_btn")):
        city_contract_btn.pressed.connect(_on_city_contract_btn)
    if city_info_btn and not city_info_btn.pressed.is_connected(Callable(self, "_on_city_info_btn")):

func _refresh_city_action_bar() -> void:
    if city_action_bar == null or world == null:
        return

    city_action_bar.visible = true

    var moving := bool(world.player.get("enroute", false))
    if city_trade_btn:
        city_trade_btn.disabled = moving
    if city_move_btn:
        city_move_btn.disabled = moving
    if city_maint_btn:
        city_maint_btn.disabled = moving

    if city_maint_btn:
        var convoy_enabled := true
        if world.get("convoy_condition_enabled") != null:
            convoy_enabled = bool(world.get("convoy_condition_enabled"))
        city_maint_btn.visible = convoy_enabled

func _get_menu_panel_for_city_actions() -> Node:
    _spawn_menu_if_needed()
    if menu_win == null:
        push_warning("GameHUD: MenuPanel が生成できませんでした。")
    return menu_win

func _call_menu_panel_action(method_name: String) -> void:
    var mp := _get_menu_panel_for_city_actions()
    if mp != null and mp.has_method(method_name):
        mp.call(method_name)
    else:
        push_warning("GameHUD: MenuPanel method not found: %s" % method_name)

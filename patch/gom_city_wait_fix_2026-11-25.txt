Guild of Merchants | 都市画面の待機欄が戻らない不具合
確認日: 2026-11-25
適用元コミット: dcc4ffd59f6e3e913b0c51ca4c1b0b3b553086dc（main）
対象: ui/game_hud.gd
エンジン: Godot 4.3 / 文字コード: UTF-8（BOMなし） / 改行: LF

原因
セーブ／ロード画面の「閉じる」とロード成功時にウィンドウが hide() する経路では、
HUD が待機欄の表示条件を再評価しないため、直前の非表示状態が残っていた。
ウィンドウの visibility_changed とダイス処理の開始・終了で状態を更新する。
ダイアログの開閉時にワールドの一時停止を反映する既存処理からも、待機欄を同期する。

適用手順
1. VS Codeで ui/game_hud.gd を開く。
2. 下記8関数を関数名で検索し、各 func 宣言から関数末尾までを対応するコードで置き換える。
   「----」の区切り行と見出しは貼り付けない。隣接するコメントや変数定義はそのまま残す。
3. 保存し、Godot 4.3で起動して確認する。
   8関数の置換で検証対象の差分と一致することを確認済み。

func _on_day(_d: int) -> void:
    _refresh()
    if world and world.use_event_dice:
        if _is_rolling:
            return
        _is_rolling = true
        _update_wait_controls_state()
        await _trigger_event_rolls()
        world.finalize_day()
        _is_rolling = false
        _update_wait_controls_state()

        var mp: Control = MenuPanelScript.new()
        mp.name = "MenuPanel"
        get_tree().root.add_child(mp)
        mp.visible = false
        if mp.has_method("set_anchors_preset"):
            mp.call("set_anchors_preset", Control.PRESET_TOP_LEFT)
        mp.world = world
        mp.hud = self
        menu_win = mp
        mp.visibility_changed.connect(_on_popup_visibility_changed)

func _spawn_trade_if_needed() -> void:

        var tw: Window = TradeWindow.new()
        tw.name = "TradeWindow"

        tw.hide()
        tw.close_requested.connect(func(): tw.hide())
        trade_win = tw
        tw.visibility_changed.connect(_on_popup_visibility_changed)

func _open_inventory_window() -> void:

    inventory_window.unresizable = false # 修正: unresizable -> resizable
    inventory_window.hide()
    if inventory_window.has_method("set"):
        inventory_window.set("world", world)
    var main := get_window()
    if main:
        main.add_child(inventory_window)
    else:
        get_tree().root.add_child(inventory_window)
    inventory_window.visibility_changed.connect(_on_popup_visibility_changed)
    _size_and_center_window(inventory_window)
    inventory_window.show()
    inventory_window.grab_focus()
    inventory_window.close_requested.connect(func():
        if is_instance_valid(inventory_window):
            inventory_window.hide()
            inventory_window.queue_free()
        inventory_window = null
    )

6. _sync_pause_state
----------------------------------------------------------------------------
func _sync_pause_state() -> void:
    if world == null:
        return
    var any := _any_popup_visible() or _any_supply_dialog_visible()
    if any:
        _popup_paused = true
        world.pause()
    else:
        _popup_paused = false
        if _user_paused:
            world.pause()
        else:
            world.resume()
    _update_play_button()
    _update_wait_controls_state()

func _open_save_load_window() -> void:
    if is_instance_valid(_save_load_win):
        _size_and_center_window(_save_load_win)
        _save_load_win.show()
        _save_load_win.popup_centered()
        _save_load_win.grab_focus()
        if _save_load_win.has_method("refresh"):
            _save_load_win.call("refresh")
        return

    var SaveLoadScript: Script = preload("res://ui/save_load_window.gd")
    var win: Window = SaveLoadScript.new()
    win.name = "SaveLoadWindow"
    win.title = "Save / Load"
    win.unresizable = false
    win.transient = true
    win.exclusive = true
    win.hide()

    var main := get_window()
    if main:
        main.add_child(win)
    else:
        get_tree().root.add_child(win)

    if win.has_method("setup"):
        win.call("setup", world)

    # 画面内の閉じるボタンやロード成功も visibility_changed で拾う。
    _save_load_win = win
    win.visibility_changed.connect(_on_popup_visibility_changed)
    win.close_requested.connect(win.hide)
    _size_and_center_window(win)
    win.popup_centered()
    win.grab_focus()

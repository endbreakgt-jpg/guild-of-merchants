# 目安: convoy_condition のヘルパー群（_tick_player_convoy_condition_on_travel_day 付近）に追記

func _apply_convoy_condition_to_layers(layers: Dictionary) -> Dictionary:
    if not convoy_condition_enabled:
        return layers

    _ensure_player_convoy_state()
    var cond: int = get_convoy_condition()
    var tier: int = _convoy_condition_tier(cond)

    # 70..100=0 / 40..69=1 / 20..39=2 / 0..19=3
    # 盗賊(bandit)には影響させず、terrain/water/weather のみ悪化させる
    var add: float = 0.0
    if tier == 1:
        add = 0.10
    elif tier == 2:
        add = 0.20
    elif tier == 3:
        add = 0.30

    if add <= 0.0:
        return layers

    var out: Dictionary = layers.duplicate(true)
    for k in ["terrain", "water", "weather"]:
        out[k] = clamp(float(out.get(k, 0.0)) + add, 0.0, 1.0)
    return out

# 2) 変更：resolve_travel_with_roll（TravelDiceV2）の置き換え
# ============================================================
# world.gd 内の同名関数を丸ごと置き換え

func resolve_travel_with_roll(roll: int, q: float = -1.0) -> void:
    if not bool(player.get("enroute", false)):
        return

    var origin := String(player.get("city", ""))
    var dest := String(player.get("dest", ""))
    var key := _route_key(origin, dest)

    var layers: Dictionary = _get_current_route_layers(key)
    layers["weather"] = _effective_weather_layer(layers)

    # escort: 盗賊遭遇を減らす（bandit layer を直接下げる）
    var escort_level: int = int(player.get("escort_level", 0))
    layers = _apply_escort_to_layers(layers, escort_level)

    # convoy: 盗賊以外の事故リスクを押し上げる（terrain/water/weatherのみ）
    layers = _apply_convoy_condition_to_layers(layers)

    # misc: 常時1.0だと misc イベントが過剰に選ばれるので、
    # 危険が低いほど misc が出やすい「控えめベース」にする
    var hazard_strength: float = _hazard_strength_for_travel_dice(layers)  # 0..1（miscは含まれない）
    layers["misc"] = clamp(0.15 + (1.0 - hazard_strength) * 0.25, 0.15, 0.40)

    var tier: String = _travel_tier_from_roll_and_layers(roll, layers)

    if log_event_dice_verbose:
        var s: int = _travel_bad_width_from_layers(layers)
        var wt := get_weather_today()
        _dice_debug("TravelDiceV2: roll=%02d tier=%s S=%d weather=%s(%d) escort=%d convoy=%d" % [
            roll, tier, s, String(wt.get("name_ja","?")), int(wt.get("severity",0)),
            escort_level, get_convoy_condition()
        ])

    if tier == "normal":
        return

    var ev: Dictionary = _pick_travel_event_for_tier_layers(tier, layers)
    if ev.is_empty():
        if tier == "great":
            ev = _pick_travel_event_for_tier_layers("good", layers)
        elif tier == "worst":
            ev = _pick_travel_event_for_tier_layers("bad", layers)
        elif tier == "misc":
            ev = _pick_travel_event_for_tier_layers("bad", layers)

    if ev.is_empty():
        return

    _apply_travel_outcome(ev)
